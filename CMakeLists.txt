cmake_minimum_required(VERSION 3.20)
project(NLite LANGUAGES CXX RC)

set(NLITE_VERSION "0.2.15" CACHE STRING "N-Lite application version")
set(NLITE_CLEANER_VERSION "5" CACHE STRING "Protected cleaner helper version")

add_executable(N-Lite WIN32 src/main.cpp src/startup_manager.cpp)
set(NLITE_APP_ICON "${CMAKE_CURRENT_SOURCE_DIR}/src/app.ico")
file(TO_CMAKE_PATH "${NLITE_APP_ICON}" NLITE_APP_ICON_RC)
configure_file(src/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app_resources.rc" @ONLY)
target_sources(N-Lite PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app_resources.rc")
target_compile_features(N-Lite PRIVATE cxx_std_17)
target_compile_definitions(N-Lite PRIVATE UNICODE _UNICODE NLITE_VERSION="${NLITE_VERSION}" NLITE_CLEANER_VERSION="${NLITE_CLEANER_VERSION}")
target_link_libraries(N-Lite PRIVATE psapi shell32 advapi32 comctl32 comdlg32 winmm dwmapi shlwapi winhttp ole32 oleaut32 taskschd)
set_property(TARGET N-Lite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(MSVC)
  target_compile_options(N-Lite PRIVATE /W4 /permissive- /utf-8 /O1 /GL /Gw /Gy)
  target_link_options(N-Lite PRIVATE /OPT:REF /OPT:ICF /LTCG)
elseif(MINGW)
  target_link_options(N-Lite PRIVATE -municode)
endif()

add_executable(N-Lite-Cleaner WIN32 src/cleaner.cpp)
target_compile_features(N-Lite-Cleaner PRIVATE cxx_std_17)
target_compile_definitions(N-Lite-Cleaner PRIVATE UNICODE _UNICODE NLITE_CLEANER_VERSION="${NLITE_CLEANER_VERSION}")
target_include_directories(N-Lite-Cleaner PRIVATE src)
target_link_libraries(N-Lite-Cleaner PRIVATE advapi32 shell32 ole32 oleaut32 taskschd)
if(MSVC)
  target_compile_options(N-Lite-Cleaner PRIVATE /W4 /permissive- /utf-8 /O1 /GL /Gw /Gy)
  target_link_options(N-Lite-Cleaner PRIVATE /OPT:REF /OPT:ICF /LTCG)
elseif(MINGW)
  target_link_options(N-Lite-Cleaner PRIVATE -municode)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(N-Lite-logic-tests tests/core_logic_tests.cpp)
  target_compile_features(N-Lite-logic-tests PRIVATE cxx_std_17)
  target_include_directories(N-Lite-logic-tests PRIVATE src)
  add_test(NAME N-Lite-logic-tests COMMAND N-Lite-logic-tests)

  add_executable(N-Lite-startup-tests tests/startup_manager_windows_tests.cpp src/startup_manager.cpp)
  target_compile_features(N-Lite-startup-tests PRIVATE cxx_std_17)
  target_include_directories(N-Lite-startup-tests PRIVATE src)
  target_link_libraries(N-Lite-startup-tests PRIVATE advapi32 shell32 comdlg32 ole32 oleaut32 taskschd)
  add_test(NAME N-Lite-startup-tests COMMAND N-Lite-startup-tests)
endif()

if(BUILD_TESTING)
  add_executable(N-Lite-cleaner-integration tests/cleaner_windows_integration.cpp src/startup_manager.cpp)
  target_compile_features(N-Lite-cleaner-integration PRIVATE cxx_std_17)
  target_compile_definitions(N-Lite-cleaner-integration PRIVATE UNICODE _UNICODE NLITE_STARTUP_TESTING NLITE_VERSION="${NLITE_VERSION}" NLITE_CLEANER_VERSION="${NLITE_CLEANER_VERSION}")
  target_include_directories(N-Lite-cleaner-integration PRIVATE src)
  target_link_libraries(N-Lite-cleaner-integration PRIVATE psapi shell32 advapi32 comctl32 comdlg32 winmm dwmapi shlwapi winhttp ole32 oleaut32 taskschd)
  add_dependencies(N-Lite-cleaner-integration N-Lite-Cleaner)
  add_test(NAME N-Lite-cleaner-integration COMMAND N-Lite-cleaner-integration)
  set_tests_properties(N-Lite-cleaner-integration PROPERTIES TIMEOUT 240)
  add_test(NAME N-Lite-startup-ui-integration COMMAND N-Lite-cleaner-integration --startup-ui-test)
  set_tests_properties(N-Lite-startup-ui-integration PROPERTIES TIMEOUT 60)
endif()
